# 文件树文件操作与拖拽重组 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 `md-reader` 左侧文件树补齐新建文档、新建文件夹、文档右键菜单、以及“仅文档拖入文件夹”的真实路径移动能力，并保证 bridge、工作区会话、自动保存和恢复链路一致。

**Architecture:** 本轮实现继续沿用当前 `bridge -> App -> WorkspaceLayout -> file-tree` 的单向数据链路，不引入额外排序真相层。`server/project-node-operations.mjs` 承担文件系统级真相与安全约束，`src/workspace/local-bridge-access.ts` 负责客户端 API 封装，新增 `src/workspace/document-path-mutation.ts` 承担纯路径改写 helper，`src/App.tsx` 负责 autosave 冻结、文件操作调度和本地状态落盘，`src/app/WorkspaceLayout.tsx` 负责搜索区按钮、目录上下文菜单和文档拖拽 UI。

**Tech Stack:** Node.js file system APIs, React 19, TypeScript, Vitest, Testing Library, plain CSS in `src/index.css`

### 1. 总体目标和范围

本实施计划基于 [2026-07-07-md-reader-文件树文件操作与拖拽重组方案.md](/C:/Code/md-reader/docs/方案/2026-07-07-md-reader-文件树文件操作与拖拽重组方案.md:1)，目标是在当前 `md-reader` 本地 bridge 工作区模型上落地以下能力：

- 搜索栏右侧新增 `新建文档` / `新建文件夹`
- 文件节点右键菜单支持 `拷贝链接`、`创建副本`、`重命名`、`删除`
- 目录节点右键菜单支持 `新建文档`、`新建文件夹`、`拷贝路径`
- 文档支持拖入文件夹做真实路径移动
- 路径变化后，当前文档、打开标签、本地恢复状态、自动保存和草稿链路保持一致

本计划明确不包含：

- 同目录内人工排序
- 文件夹拖拽
- 目录 rename / delete / duplicate
- 递归批量移动、批量删除、撤销、回收站
- deep-link 协议

import path from 'node:path'
import { copyFile, mkdir, rename, rm, stat, writeFile } from 'node:fs/promises'

import { resolveMarkdownDocumentPath } from './project-reader.mjs'

export class ProjectNodeOperationError extends Error {
  constructor(code, message) {
    super(message)
    this.name = 'ProjectNodeOperationError'
    this.code = code
  }
}

function ensureMarkdownPath(targetPath) {
  if (!targetPath.endsWith('.md') && !targetPath.endsWith('.mdx')) {
    throw new ProjectNodeOperationError('INVALID_TARGET_PATH', `Markdown path required: ${targetPath}`)
  }
}

function resolveMarkdownNodePath(projectRoot, contentRoots, relativePath) {
  return resolveMarkdownDocumentPath(projectRoot, contentRoots, relativePath)
}

function resolveDirectoryNodePath(projectRoot, contentRoots, relativePath) {
  const normalizedPath = String(relativePath ?? '').trim().replaceAll('\\', '/')
  if (!normalizedPath) {
    throw new ProjectNodeOperationError('INVALID_TARGET_PATH', 'Directory path required')
  }

  const root = path.resolve(projectRoot)
  const resolvedNodePath = path.resolve(root, normalizedPath)
  const allowedRoots = contentRoots.map((contentRoot) => path.resolve(root, contentRoot))
  const isAllowed = allowedRoots.some(
    (allowedRoot) =>
      resolvedNodePath === allowedRoot ||
      resolvedNodePath.startsWith(`${allowedRoot}${path.sep}`),
  )

  if (!isAllowed) {
    throw new ProjectNodeOperationError(
      'INVALID_TARGET_PATH',
      `Directory path is outside allowed content roots: ${relativePath}`,
    )
  }

  return {
    normalizedPath,
    resolvedNodePath,
  }
}

function resolveDirectoryNodePath(projectRoot, contentRoots, relativePath) {
  const normalizedPath = String(relativePath ?? '').trim().replaceAll('\\', '/')
  if (!normalizedPath) {
    throw new ProjectNodeOperationError('INVALID_TARGET_PATH', 'Directory path required')
  }

  const root = path.resolve(projectRoot)
  const resolvedNodePath = path.resolve(root, normalizedPath)
  const allowedRoots = contentRoots.map((contentRoot) => path.resolve(root, contentRoot))
  const isAllowed = allowedRoots.some(
    (allowedRoot) =>
      resolvedNodePath === allowedRoot ||
      resolvedNodePath.startsWith(`${allowedRoot}${path.sep}`),
  )

  if (!isAllowed) {
    throw new ProjectNodeOperationError('INVALID_TARGET_PATH', `Directory path is outside allowed content roots: ${relativePath}`)
  }

  return {
    normalizedPath,
    resolvedNodePath,
  }
}

export async function createMarkdownDocument(projectRoot, contentRoots, documentPath, content = '') {
  ensureMarkdownPath(documentPath)
  const { normalizedPath, resolvedDocumentPath } = resolveMarkdownNodePath(projectRoot, contentRoots, documentPath)
  await mkdir(path.dirname(resolvedDocumentPath), { recursive: true })
  await writeFile(resolvedDocumentPath, content, { flag: 'wx', encoding: 'utf8' })
  const metadata = await stat(resolvedDocumentPath)
  return { kind: 'file', path: normalizedPath, mtimeMs: metadata.mtimeMs }
}
```

export interface BridgeNodeMutationPayload {
  ok: true
  kind: 'file' | 'directory'
  path: string
  nextPath?: string

export async function createDocumentNodeFromBridge(
  projectId: string,
  profileId: string,
  documentPath: string,
  content: string,
  options: FetchOptions = {},
): Promise<BridgeNodeMutationPayload> {
  const fetchImpl = getFetch(options.fetchImpl)
  const baseUrl = getBaseUrl(options.baseUrl)
  const response = await fetchImpl(
    `${baseUrl}/api/projects/${encodeURIComponent(projectId)}/nodes/create-document?profileId=${encodeURIComponent(profileId)}`,
    {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ path: documentPath, content }),
    },
  )

  return readJsonResponse<BridgeNodeMutationPayload>(response)
}

export function rewriteExactPathRecord<T>(record: Record<string, T>, oldPath: string, nextPath: string): Record<string, T> {
  if (!(oldPath in record)) {
    return record
  }

  const { [oldPath]: value, ...rest } = record
  return {
    ...rest,
    [nextPath]: value,
  }
}

export function rewriteLocalStateDocumentPath(
  state: WorkspaceLocalState,
  oldPath: string,
  nextPath: string,
): WorkspaceLocalState {
  return {
    ...state,
    openDocumentPaths: state.openDocumentPaths.map((path) => (path === oldPath ? nextPath : path)),
    activeDocumentPath: state.activeDocumentPath === oldPath ? nextPath : state.activeDocumentPath,
    tabStateByDocument: rewriteExactPathRecord(state.tabStateByDocument, oldPath, nextPath),
    readingProgressByDocument: rewriteExactPathRecord(state.readingProgressByDocument, oldPath, nextPath),
  }
}
```

function resolveDefaultDirectoryPath(overrideDirectoryPath?: string): string {
  if (overrideDirectoryPath) {
